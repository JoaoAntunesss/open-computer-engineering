Uma `struct` junta vários valores relacionados num só tipo — o equivalente mais próximo de uma classe sem métodos. Combinando structs com apontadores e memória dinâmica, constróis as estruturas de dados clássicas. A primeira é a **lista ligada**.

## Structs

```c
// structs.c
#include <stdio.h>
#include <string.h>

typedef struct {
    char nome[40];
    int numero;
    double media;
} Aluno;

void mostra(const Aluno *a) {               // passa o endereço: não copia a struct
    printf("%-10s %6d %5.1f\n", a->nome, a->numero, a->media);
}

int main(void) {
    Aluno a = {"Ana", 2400101, 16.4};
    Aluno b = a;                             // structs copiam-se com = (ao contrário dos vetores)
    strcpy(b.nome, "Bruno");
    b.media = 13.0;
    Aluno turma[] = {a, b, {.nome = "Carla", .numero = 2400215, .media = 18.2}};
    for (int i = 0; i < 3; i++) mostra(&turma[i]);
    printf("sizeof(Aluno) = %zu\n", sizeof(Aluno));
    return 0;
}
```

- `a.campo` acede a um campo; `p->campo` é atalho para `(*p).campo` quando `p` é um apontador.
- `typedef` dá um nome curto ao tipo.
- Para structs grandes, passa um `const T *` às funções, para não copiar tudo.
- O `sizeof` pode ser maior do que a soma dos campos: o compilador acrescenta **enchimento** (*padding*) para alinhar cada campo em memória. Aqui: 40 + 4 + 4 de enchimento + 8 = **56** bytes.

## Listas ligadas

Um vetor guarda os elementos lado a lado. Uma **lista ligada** guarda-os em **nós** espalhados pela memória, cada um com um apontador para o seguinte.

```text
cabeça → [3 | •] → [7 | •] → [9 | NULL]
```

Comparação:

- **Vetor**: acesso ao i-ésimo em O(1); inserir no início é O(n) (é preciso deslocar tudo).
- **Lista ligada**: acesso ao i-ésimo em O(n); inserir no início é O(1); cada nó gasta memória extra com o apontador e fica longe dos outros na cache.

## Implementação

```c
// lista.c
#include <stdio.h>
#include <stdlib.h>

typedef struct No {
    int valor;
    struct No *prox;
} No;

int insere_inicio(No **cabeca, int v) {       // ** porque altera a cabeça
    No *n = malloc(sizeof *n);
    if (!n) return 0;
    n->valor = v;
    n->prox = *cabeca;
    *cabeca = n;
    return 1;
}

int insere_ordenado(No **cabeca, int v) {
    No **p = cabeca;
    while (*p && (*p)->valor < v) p = &(*p)->prox;   // p aponta para o "elo" onde inserir
    No *n = malloc(sizeof *n);
    if (!n) return 0;
    n->valor = v;
    n->prox = *p;
    *p = n;
    return 1;
}

int remove_valor(No **cabeca, int v) {
    for (No **p = cabeca; *p; p = &(*p)->prox)
        if ((*p)->valor == v) { No *lixo = *p; *p = lixo->prox; free(lixo); return 1; }
    return 0;
}

void imprime(const No *n) {
    for (; n; n = n->prox) printf("%d -> ", n->valor);
    printf("NULL\n");
}

void liberta(No *n) {
    while (n) { No *seg = n->prox; free(n); n = seg; }
}

int main(void) {
    No *lista = NULL;
    int dados[] = {7, 3, 9, 1, 5};
    for (int i = 0; i < 5; i++) insere_ordenado(&lista, dados[i]);
    imprime(lista);                 // 1 -> 3 -> 5 -> 7 -> 9 -> NULL
    remove_valor(&lista, 5);
    insere_inicio(&lista, 0);
    imprime(lista);                 // 0 -> 1 -> 3 -> 7 -> 9 -> NULL
    liberta(lista);
    return 0;
}
```

O truque do **apontador para apontador** (`No **p`) evita tratar a cabeça como caso especial: `p` aponta sempre para o campo que vai ter de mudar — seja a variável `lista`, seja o `prox` de um nó.

Repara também no `liberta`: guarda o `prox` **antes** de libertar o nó, porque depois do `free` não se pode ler.

## Variantes

- **Duplamente ligada**: cada nó tem `prox` e `ant`; remover um nó conhecido é O(1).
- **Circular**: o último aponta para o primeiro.
- **Com cauda**: guardar também um apontador para o último nó torna "inserir no fim" O(1) — é a base de uma fila (lição 7).

## Listas genéricas

Para guardar qualquer tipo, o nó pode ter um `void *dados` e a lista recebe funções para comparar e libertar os elementos — o mesmo padrão do `qsort`. É o primeiro projeto prático desta cadeira.

## Erros comuns

- Esquecer de pôr `prox = NULL` no último nó.
- Perder a cabeça da lista ao percorrê-la com a própria variável `lista`.
- Usar um nó depois de o libertar.
- Passar `No *` a uma função que tem de alterar a cabeça (precisa de `No **`).
- Calcular o tamanho da lista a cada iteração de um ciclo (O(n²)).

// solucoes6.c
#include <stdio.h>
#include <stdlib.h>

typedef struct No { int valor; struct No *prox; } No;

int comprimento(const No *n) { int c = 0; for (; n; n = n->prox) c++; return c; }
int comprimento_rec(const No *n) { return n ? 1 + comprimento_rec(n->prox) : 0; }

No *inverte(No *n) {
    No *ant = NULL;
    while (n) { No *seg = n->prox; n->prox = ant; ant = n; n = seg; }
    return ant;
}

int tem_ciclo(const No *n) {
    const No *lenta = n, *rapida = n;
    while (rapida && rapida->prox) {
        lenta = lenta->prox;
        rapida = rapida->prox->prox;
        if (lenta == rapida) return 1;
    }
    return 0;
}

int main(void) {
    No c = {3, NULL}, b = {2, &c}, a = {1, &b};          // nós na pilha, só para testar
    printf("%d %d\n", comprimento(&a), comprimento_rec(&a));   // 3 3
    No *r = inverte(&a);
    for (No *p = r; p; p = p->prox) printf("%d ", p->valor);   // 3 2 1
    printf("\n%d\n", tem_ciclo(r));                            // 0
    a.prox = r;                                                // cria um ciclo: 1 -> 3
    printf("%d\n", tem_ciclo(r));                              // 1
    return 0;
}
