Pilhas, filas e tabelas de dispersão são as estruturas que mais vais usar, muitas vezes sem dar por isso: a pilha guarda as chamadas de funções, as filas organizam tarefas e pedidos, e as tabelas de dispersão estão por trás dos dicionários do Python. Implementá-las em C mostra o que custa cada operação.

## Pilha (LIFO)

O último a entrar é o primeiro a sair. Operações: **push** (empilhar), **pop** (desempilhar), **top** (ver o topo). Implementa-se com um vetor dinâmico e um índice do topo; todas as operações são O(1).

```c
// pilha.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct { char *d; size_t n, cap; } Pilha;

int push(Pilha *p, char c) {
    if (p->n == p->cap) {
        size_t nc = p->cap ? 2 * p->cap : 8;
        char *x = realloc(p->d, nc);
        if (!x) return 0;
        p->d = x; p->cap = nc;
    }
    p->d[p->n++] = c;
    return 1;
}
int vazia(const Pilha *p) { return p->n == 0; }
char pop(Pilha *p) { return p->d[--p->n]; }

int equilibrado(const char *s) {        // "(a[b]{c})" sim, "(]" não
    Pilha p = {0};
    int ok = 1;
    for (; *s && ok; s++) {
        if (strchr("([{", *s)) ok = push(&p, *s);
        else if (strchr(")]}", *s)) {
            char esperado = *s == ')' ? '(' : *s == ']' ? '[' : '{';
            ok = !vazia(&p) && pop(&p) == esperado;
        }
    }
    ok = ok && vazia(&p);
    free(p.d);
    return ok;
}

int main(void) {
    const char *testes[] = {"(a[b]{c})", "(]", "((", "f(x[0]) + {y}"};
    for (int i = 0; i < 4; i++) printf("%-15s %s\n", testes[i], equilibrado(testes[i]) ? "sim" : "não");
    return 0;
}
```

Outros usos: anular ações (Ctrl+Z), avaliar expressões, percorrer grafos em profundidade, e a própria pilha de chamadas do programa.

## Fila (FIFO)

O primeiro a entrar é o primeiro a sair. Operações: **enqueue** (pôr no fim) e **dequeue** (tirar do início). Com um vetor normal, tirar do início obriga a deslocar tudo (O(n)). A solução é um **buffer circular**: dois índices que dão a volta ao vetor.

```c
// fila.c
#include <stdio.h>

#define CAP 4
typedef struct { int d[CAP]; int ini, n; } Fila;

int enqueue(Fila *f, int x) {
    if (f->n == CAP) return 0;                 // cheia
    f->d[(f->ini + f->n) % CAP] = x;
    f->n++;
    return 1;
}
int dequeue(Fila *f, int *x) {
    if (f->n == 0) return 0;                   // vazia
    *x = f->d[f->ini];
    f->ini = (f->ini + 1) % CAP;
    f->n--;
    return 1;
}

int main(void) {
    Fila f = {0};
    int x;
    for (int i = 1; i <= 4; i++) enqueue(&f, i * 10);
    printf("cheia? %s\n", enqueue(&f, 50) ? "não" : "sim");   // sim
    dequeue(&f, &x); printf("%d\n", x);                       // 10
    enqueue(&f, 50);                                          // reutiliza a posição libertada
    while (dequeue(&f, &x)) printf("%d ", x);                 // 20 30 40 50
    printf("\n");
    return 0;
}
```

Buffers circulares usam-se em teclados, placas de som e de rede, e em comunicação entre processos. Uma fila com lista ligada (cabeça + cauda) também dá O(1) e não tem limite fixo.

## Tabela de dispersão

Uma **tabela de dispersão** (*hash table*) guarda pares chave → valor com procura, inserção e remoção em **O(1) em média**. A ideia:

1. Uma **função de dispersão** transforma a chave num número.
2. Esse número, módulo o tamanho da tabela, dá o índice de um "balde".
3. Chaves diferentes podem cair no mesmo balde (**colisão**). Com **encadeamento**, cada balde é uma lista ligada.

```c
// hash.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct Ent { char *chave; int valor; struct Ent *prox; } Ent;
typedef struct { Ent **baldes; size_t n_baldes, n; } Tabela;

static unsigned long djb2(const char *s) {     // função de dispersão de Dan Bernstein
    unsigned long h = 5381;
    for (; *s; s++) h = h * 33 + (unsigned char)*s;
    return h;
}

Ent *procura(Tabela *t, const char *k) {
    for (Ent *e = t->baldes[djb2(k) % t->n_baldes]; e; e = e->prox)
        if (strcmp(e->chave, k) == 0) return e;
    return NULL;
}

int poe(Tabela *t, const char *k, int v) {
    Ent *e = procura(t, k);
    if (e) { e->valor = v; return 1; }
    e = malloc(sizeof *e);
    if (!e || !(e->chave = malloc(strlen(k) + 1))) { free(e); return 0; }
    strcpy(e->chave, k);
    e->valor = v;
    size_t i = djb2(k) % t->n_baldes;
    e->prox = t->baldes[i];
    t->baldes[i] = e;
    t->n++;
    return 1;
}

void liberta(Tabela *t) {
    for (size_t i = 0; i < t->n_baldes; i++)
        for (Ent *e = t->baldes[i], *s; e; e = s) { s = e->prox; free(e->chave); free(e); }
    free(t->baldes);
}

int main(void) {
    Tabela t = {calloc(64, sizeof(Ent *)), 64, 0};
    if (!t.baldes) return 1;
    const char *texto[] = {"o", "gato", "viu", "o", "rato", "e", "o", "rato", "fugiu"};
    for (int i = 0; i < 9; i++) {
        Ent *e = procura(&t, texto[i]);
        poe(&t, texto[i], e ? e->valor + 1 : 1);
    }
    printf("o: %d, rato: %d, gato: %d, distintas: %zu\n",
           procura(&t, "o")->valor, procura(&t, "rato")->valor, procura(&t, "gato")->valor, t.n);   // 3, 2, 1, 6
    liberta(&t);
    return 0;
}
```

O desempenho depende do **fator de carga** (n / número de baldes). Quando passa de cerca de 0.75, duplica-se o número de baldes e reinserem-se todos os elementos (**rehash**). Uma boa função de dispersão espalha as chaves de forma uniforme; uma má (por exemplo, "soma dos caracteres") põe "ab" e "ba" no mesmo balde.

## Erros comuns

- Fazer `pop` numa pilha vazia (lê fora do vetor).
- Na fila circular, confundir "cheia" com "vazia" quando `ini == fim` (guardar `n` resolve).
- Guardar na tabela o apontador para a string do chamador em vez de uma cópia.
- Nunca aumentar a tabela: com muitos elementos, as listas ficam longas e a procura passa a O(n).

// rpn.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int rpn(const char *s, double *res) {
    double p[64]; int n = 0;
    char buf[128]; snprintf(buf, sizeof buf, "%s", s);
    for (char *tok = strtok(buf, " "); tok; tok = strtok(NULL, " ")) {
        if (isdigit((unsigned char)tok[0])) { if (n == 64) return 0; p[n++] = atof(tok); continue; }
        if (n < 2) return 0;
        double b = p[--n], a = p[--n];
        switch (tok[0]) {
            case '+': p[n++] = a + b; break;
            case '-': p[n++] = a - b; break;
            case '*': p[n++] = a * b; break;
            case '/': p[n++] = a / b; break;
            default: return 0;
        }
    }
    if (n != 1) return 0;
    *res = p[0];
    return 1;
}

int main(void) {
    double r;
    if (rpn("3 4 + 2 *", &r)) printf("%g\n", r);    // 14
    if (rpn("5 1 2 + 4 * + 3 -", &r)) printf("%g\n", r);   // 14
    return 0;
}
